const fetchFromGemini = async (prompt, apiKey) => {
  if (!apiKey) return '{"error": "Please add your Gemini API Key at the top of the code to use AI features."}';
  const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
  const payload = { contents: [{ parts: [{ text: prompt }] }] };
  let retries = 5;
  let delays = [1000, 2000, 4000, 8000, 16000];

  for (let i = 0; i <= retries; i++) {
    try {
      const response = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      
      if (!response.ok) {
         if (response.status === 400 || response.status === 401 || response.status === 403) {
             return '{"error": "Invalid API Key or Unauthorized. Please check your API key."}';
         }
         throw new Error(`HTTP error! status: ${response.status}`);
      }
      
      const data = await response.json();
      return data.candidates?.[0]?.content?.parts?.[0]?.text || '{"error": "No response generated."}';
    } catch (error) {
      if (i === retries) {
        return '{"error": "Connection failed. Please check your network or API Key."}';
      }
      await new Promise(res => setTimeout(res, delays[i]));
    }
  }
};

const parseGeminiJSON = (text) => {
  try {
    const start = text.indexOf('{');
    const end = text.lastIndexOf('}');
    if (start !== -1 && end !== -1) {
      return JSON.parse(text.slice(start, end + 1));
    }
    return null;
  } catch (e) {
    return null;
  }
};

const cleanFallbackText = (text) => {
  if (!text) return "";

  return text.split(marker + 'json').join('').split(marker).join('').trim();
};

const safeRender = (val) => {
  if (val === null || val === undefined) return '';
  if (typeof val === 'string' || typeof val === 'number' || typeof val === 'boolean') return String(val);
  if (typeof val === 'object') return JSON.stringify(val);
  return String(val);
};

    <div className="space-y-10 animate-fadeIn">
      <div className="relative overflow-hidden bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-800 rounded-3xl p-8 md:p-12 shadow-2xl transition-colors duration-300">
        <div className="absolute top-0 left-0 w-full h-full bg-gradient-to-br from-blue-600/10 via-purple-600/5 to-transparent pointer-events-none"></div>
        <div className="absolute -top-24 -left-24 w-64 h-64 bg-blue-600/20 rounded-full blur-3xl pointer-events-none"></div>
        <div className="absolute -bottom-24 -right-24 w-64 h-64 bg-purple-600/20 rounded-full blur-3xl pointer-events-none"></div>
        
        <div className="relative z-10 flex flex-col md:flex-row items-center justify-between gap-8">
          <div className="flex-1 text-center md:text-start">
            <h2 className="text-4xl md:text-5xl font-extrabold mb-4 text-gray-900 dark:text-white leading-tight">
              {t.welcome} <br/><span className="bg-clip-text text-transparent bg-gradient-to-r from-blue-500 to-purple-500">{t.currentLvl} {userLevel}</span>
            </h2>
            <p className="text-gray-600 dark:text-gray-400 text-lg mb-8 max-w-xl mx-auto md:mx-0 leading-relaxed">
              {t.desc}
            </p>
            <div className="flex flex-wrap justify-center md:justify-start gap-4">
              <button onClick={() => setTab('plan')} className="bg-gradient-to-r from-blue-600 to-blue-500 text-white px-8 py-3.5 rounded-xl font-bold hover:shadow-[0_0_20px_rgba(37,99,235,0.4)] transition-all flex items-center gap-2">
                <Calendar className="w-5 h-5" /> {t.createPlan}
              </button>
            </div>
          </div>
        </div>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        <div className="bg-gradient-to-br from-indigo-100 dark:from-indigo-900/40 to-blue-50 dark:to-blue-900/20 p-6 rounded-3xl border border-indigo-200 dark:border-indigo-500/30 flex flex-col justify-center items-center text-center relative overflow-hidden group transition-colors">
          <Quote className="absolute top-4 left-4 w-12 h-12 text-indigo-500/10 -rotate-12" />
          <h3 className="text-lg font-bold text-indigo-700 dark:text-indigo-300 mb-4 flex items-center gap-2">
            <Sparkles className="w-4 h-4" /> {t.quoteTitle}
          </h3>
          {loadingInsight ? (
            <Activity className="w-8 h-8 text-indigo-500 dark:text-indigo-400 animate-spin" />
          ) : (
            <>
              <p className="text-xl font-serif italic text-gray-800 dark:text-white mb-2" dir="ltr">"{safeRender(insight?.quote)}"</p>
              <p className="text-sm text-indigo-600 dark:text-indigo-300/80">{safeRender(insight?.translated_quote)}</p>
            </>
          )}
        </div>

        <div className="bg-gradient-to-br from-amber-100 dark:from-amber-900/40 to-orange-50 dark:to-orange-900/20 p-6 rounded-3xl border border-amber-200 dark:border-amber-500/30 flex flex-col justify-center items-center text-center group transition-colors">
          <h3 className="text-lg font-bold text-amber-700 dark:text-amber-300 mb-4 flex items-center gap-2">
             <Zap className="w-4 h-4" /> {t.tipTitle}
          </h3>
          {loadingInsight ? (
            <Activity className="w-8 h-8 text-amber-500 dark:text-amber-400 animate-spin" />
          ) : (
            <>
              <p className="text-lg text-gray-800 dark:text-white font-medium" dir="ltr">{safeRender(insight?.pro_tip)}</p>
              <p className="text-sm text-amber-600 dark:text-amber-300/80 mt-2">{safeRender(insight?.translated_tip)}</p>
            </>
          )}
        </div>
      </div>

      <div>
        <h3 className="text-2xl font-bold text-gray-900 dark:text-white mb-6 flex items-center gap-2"><Activity className="w-6 h-6 text-purple-500 dark:text-purple-400"/> {t.learningPaths}</h3>
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
          {t.paths.map(path => (
             <DashboardCard key={path.id} {...path} onClick={() => setTab(path.id)} />
          ))}
        </div>
      </div>
    </div>
  );
}

function StudyPlan({ userLevel, setUserLevel, lang, globalVocab }) {
  const [targetLevel, setTargetLevel] = useState('C1');
  const [targetDate, setTargetDate] = useState('');
  const [hours, setHours] = useState(1);
  const [minutes, setMinutes] = useState(30);
  
  const [planData, setPlanData] = useState(null);
  const [currentWeek, setCurrentWeek] = useState(1);
  const [vocabModal, setVocabModal] = useState(null);
  const [grammarModal, setGrammarModal] = useState(null);

  const t = {
    title: lang === 'ar' ? 'إعداد خطة المذاكرة والمستوى' : 'Set Up Study Plan & Level',
    currentLvl: lang === 'ar' ? 'مستواك الحالي' : 'Current Level',
    targetLvl: lang === 'ar' ? 'المستوى المرغوب' : 'Target Level',
    dateLabel: lang === 'ar' ? 'تاريخ الوصول للهدف' : 'Target Date',
    timeLabel: lang === 'ar' ? 'وقت المذاكرة اليومي' : 'Daily Study Time',
    hPlaceholder: lang === 'ar' ? 'ساعات' : 'Hours',
    mPlaceholder: lang === 'ar' ? 'دقائق' : 'Mins',
    gen: lang === 'ar' ? 'توليد الجدول الأسبوعي' : 'Generate Weekly Schedule',
    week: lang === 'ar' ? 'الأسبوع' : 'Week',
    day: lang === 'ar' ? 'اليوم' : 'Day',
    reset: lang === 'ar' ? 'إعادة ضبط الخطة' : 'Reset Plan',
    exportPdf: lang === 'ar' ? 'تصدير الخطة كـ PDF' : 'Export as PDF',
    activity: lang === 'ar' ? 'النشاط' : 'Activity',
    summary: lang === 'ar' ? 'المجموع' : 'Summary',
    words: lang === 'ar' ? 'كلمات' : 'words',
    vocabHeader: lang === 'ar' ? 'كلمات اليوم' : 'Daily Vocabulary',
    learnTarget: lang === 'ar' ? '10 كلمات للتعلم' : '10 words to learn',
    grammarHeader: lang === 'ar' ? 'قاعدة اليوم' : 'Daily Grammar',
    grammarTarget: lang === 'ar' ? 'القاعدة المطلوبة دراستها' : 'Rule to study',
    alertTime: lang === 'ar' ? 'يرجى تحديد وقت المذاكرة!' : 'Please specify study time!',
    alertDate: lang === 'ar' ? 'يرجى تحديد تاريخ الهدف بشكل صحيح!' : 'Please select a valid target date!',
    categories: {
      listen: lang === 'ar' ? 'استماع' : 'Listening',
      read: lang === 'ar' ? 'قراءة' : 'Reading',
      speak: lang === 'ar' ? 'محادثة' : 'Speaking',
      write: lang === 'ar' ? 'كتابة' : 'Writing',
      vocab: lang === 'ar' ? 'مفردات' : 'Vocabulary',
      grammar: lang === 'ar' ? 'قواعد' : 'Grammar',
      total: lang === 'ar' ? 'المجموع اليومي' : 'Daily Total'
    }
  };

  const wordsPerDay = 10;

  const suitableVocab = useMemo(() => {
    const getLevelWeight = (l) => {
      if (!l) return 0;
      const str = l.toUpperCase();
      if (str.includes('A1')) return 1;
      if (str.includes('A2')) return 2;
      if (str.includes('B1')) return 3;
      if (str.includes('B2')) return 4;
      if (str.includes('C1')) return 5;
      if (str.includes('C2')) return 6;
      return 0;
    };
    
    const userWeight = getLevelWeight(userLevel) || 1; 
    const filtered = globalVocab.filter(w => {
      const wWeight = getLevelWeight(w.level);
      return wWeight >= userWeight || wWeight === 0; 
    });
    
    return filtered.length >= wordsPerDay ? filtered : globalVocab;
  }, [globalVocab, userLevel]);

  const relevantGrammarTopics = useMemo(() => {
    const levelsToInclude = GRAMMAR_LEVELS.filter(g => {
        return g.level.includes(userLevel) || g.level.includes(targetLevel);
    });
    if (levelsToInclude.length === 0) return GRAMMAR_LEVELS[2].topics; 
    return levelsToInclude.flatMap(g => g.topics);
  }, [userLevel, targetLevel]);

  const generatePlan = () => {
    const totalMins = (parseInt(hours) || 0) * 60 + (parseInt(minutes) || 0);
    if (totalMins <= 0) {
      alert(t.alertTime);
      return;
    }
    if (!targetDate) {
      alert(t.alertDate);
      return;
    }

    let calculatedWeeks = 4; 
    const start = new Date();
    const end = new Date(targetDate);
    if (end > start) {
      const diffTime = Math.abs(end - start);
      const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
      calculatedWeeks = Math.max(1, Math.ceil(diffDays / 7));
    } else {
      alert(t.alertDate);
      return;
    }

    const catData = [
      { id: 'listen', name: t.categories.listen, icon: Headphones, mins: Math.round(totalMins * 0.35), headerBg: 'bg-blue-100 dark:bg-blue-900/30', cellBg: 'bg-blue-50 dark:bg-blue-900/10', textColor: 'text-blue-700 dark:text-blue-400' },
      { id: 'read', name: t.categories.read, icon: BookText, mins: Math.round(totalMins * 0.25), headerBg: 'bg-emerald-100 dark:bg-emerald-900/30', cellBg: 'bg-emerald-50 dark:bg-emerald-900/10', textColor: 'text-emerald-700 dark:text-emerald-400' },
      { id: 'speak', name: t.categories.speak, icon: MessageSquare, mins: Math.round(totalMins * 0.15), headerBg: 'bg-pink-100 dark:bg-pink-900/30', cellBg: 'bg-pink-50 dark:bg-pink-900/10', textColor: 'text-pink-700 dark:text-pink-400' },
      { id: 'write', name: t.categories.write, icon: PenTool, mins: Math.round(totalMins * 0.10), headerBg: 'bg-indigo-100 dark:bg-indigo-900/30', cellBg: 'bg-indigo-50 dark:bg-indigo-900/10', textColor: 'text-indigo-700 dark:text-indigo-400' },
      { id: 'vocab', name: t.categories.vocab, icon: BookOpen, mins: Math.round(totalMins * 0.10), headerBg: 'bg-fuchsia-100 dark:bg-fuchsia-900/30', cellBg: 'bg-fuchsia-50 dark:bg-fuchsia-900/10', textColor: 'text-fuchsia-700 dark:text-fuchsia-400' },
      { id: 'grammar', name: t.categories.grammar, icon: FileText, mins: Math.round(totalMins * 0.05), headerBg: 'bg-orange-100 dark:bg-orange-900/30', cellBg: 'bg-orange-50 dark:bg-orange-900/10', textColor: 'text-orange-700 dark:text-orange-400' },
    ];
    setPlanData({ totalWeeks: calculatedWeeks, categories: catData });
    setCurrentWeek(1);
  };

  const handleNextWeek = () => {
    if (currentWeek < planData.totalWeeks) setCurrentWeek(prev => prev + 1);
  };

  const handlePrevWeek = () => {
    if (currentWeek > 1) setCurrentWeek(prev => prev - 1);
  };

  const openVocabModal = (dayIndex) => {
    const startIdx = ((currentWeek - 1) * 7 + dayIndex) * wordsPerDay;
    const dayWords = [];
    
    for (let i = 0; i < wordsPerDay; i++) {
       if(suitableVocab.length === 0) break;
       const globalIdx = (startIdx + i) % suitableVocab.length;
       dayWords.push(suitableVocab[globalIdx]);
    }
    
    setVocabModal({ day: dayIndex + 1, words: dayWords });
  };

  const openGrammarModal = (dayIndex) => {
    const globalIdx = ((currentWeek - 1) * 7 + dayIndex);
    const topic = relevantGrammarTopics[globalIdx % relevantGrammarTopics.length];
    setGrammarModal({ day: dayIndex + 1, topic });
  };

  const handlePrint = () => {
    window.print();
  };

  const formatTime = (mins) => {
    const h = Math.floor(mins / 60);
    const m = mins % 60;
    if (h > 0 && m > 0) return `${h}h ${m}m`;
    if (h > 0) return `${h}h`;
    return `${m}m`;
  };

  const dailyTotalMins = planData?.categories.reduce((acc, cat) => acc + cat.mins, 0) || 0;
  const weeklyTotalMins = dailyTotalMins * 7;

  const todayStr = new Date().toISOString().split('T')[0];

  return (
    <div className="space-y-8 animate-fadeIn print:m-0 print:p-0">
      {!planData && (
        <div className="bg-white dark:bg-[#121214] p-8 md:p-12 rounded-3xl border border-gray-200 dark:border-gray-800 shadow-2xl relative overflow-hidden transition-colors">
          <div className="absolute top-0 right-0 w-64 h-64 bg-blue-600/10 rounded-full blur-3xl pointer-events-none"></div>
          
          <h2 className="text-3xl font-extrabold mb-8 text-gray-900 dark:text-white flex items-center gap-3 relative z-10">
            <Sparkles className="text-blue-600 dark:text-blue-400 w-8 h-8" /> {t.title}
          </h2>
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6 mb-10 relative z-10">
            <div className="bg-gray-50 dark:bg-[#1a1a1e] p-6 rounded-2xl border border-gray-200 dark:border-gray-700/50 flex flex-col justify-center">
              <label className="block text-sm text-gray-500 dark:text-gray-400 mb-2 font-semibold">{t.currentLvl}</label>
              <select value={userLevel} onChange={(e) => setUserLevel(e.target.value)} className="w-full bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-700 rounded-xl px-4 py-3 text-gray-900 dark:text-white font-bold focus:outline-none focus:border-blue-500 cursor-pointer">
                {['A1', 'A2', 'B1', 'B2', 'C1', 'C2'].map(l => (
                  <option key={l} value={l}>{l}</option>
                ))}
              </select>
            </div>
            
            <div className="bg-gray-50 dark:bg-[#1a1a1e] p-6 rounded-2xl border border-gray-200 dark:border-gray-700/50 flex flex-col justify-center">
              <label className="block text-sm text-gray-500 dark:text-gray-400 mb-2 font-semibold">{t.targetLvl}</label>
              <select value={targetLevel} onChange={(e) => setTargetLevel(e.target.value)} className="w-full bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-700 rounded-xl px-4 py-3 text-gray-900 dark:text-white font-bold focus:outline-none focus:border-blue-500 cursor-pointer">
                {['A2', 'B1', 'B2', 'C1', 'C2'].map(l => (
                  <option key={l} value={l}>{l}</option>
                ))}
              </select>
            </div>

            <div className="bg-gray-50 dark:bg-[#1a1a1e] p-6 rounded-2xl border border-gray-200 dark:border-gray-700/50 flex flex-col justify-center">
              <label className="block text-sm text-gray-500 dark:text-gray-400 mb-2 font-semibold">{t.dateLabel}</label>
              <input type="date" min={todayStr} value={targetDate} onChange={(e) => setTargetDate(e.target.value)} className="w-full bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-700 rounded-xl px-4 py-3 text-gray-900 dark:text-white font-bold focus:outline-none focus:border-blue-500" dir="ltr" />
            </div>

            <div className="bg-gray-50 dark:bg-[#1a1a1e] p-6 rounded-2xl border border-gray-200 dark:border-gray-700/50 flex flex-col justify-center">
              <label className="block text-sm text-gray-500 dark:text-gray-400 mb-2 font-semibold">{t.timeLabel}</label>
              <div className="flex gap-2">
                <div className="flex-1">
                   <input type="number" min="0" max="24" value={hours} onChange={(e) => setHours(e.target.value)} placeholder={t.hPlaceholder} className="w-full bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-700 rounded-xl px-3 py-3 text-gray-900 dark:text-white font-bold text-center focus:outline-none focus:border-blue-500" />
                   <span className="text-xs text-gray-500 flex justify-center mt-1">{t.hPlaceholder}</span>
                </div>
                <div className="flex-1">
                   <input type="number" min="0" max="59" value={minutes} onChange={(e) => setMinutes(e.target.value)} placeholder={t.mPlaceholder} className="w-full bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-700 rounded-xl px-3 py-3 text-gray-900 dark:text-white font-bold text-center focus:outline-none focus:border-blue-500" />
                   <span className="text-xs text-gray-500 flex justify-center mt-1">{t.mPlaceholder}</span>
                </div>
              </div>
            </div>
          </div>
          <button onClick={generatePlan} className="w-full bg-gradient-to-r from-blue-600 to-blue-500 text-white font-extrabold text-lg py-5 rounded-2xl transition-all hover:shadow-lg">{t.gen}</button>
        </div>
      )}

      {planData && (
        <div className="bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-800 rounded-3xl shadow-2xl p-6 md:p-8 animate-fadeIn transition-colors relative print:shadow-none print:border-none print:p-0">
          <div className="flex flex-col md:flex-row justify-between items-center mb-8 gap-4 print:hidden">
             <div className="flex items-center gap-4 bg-gray-50 dark:bg-[#1a1a1e] px-4 py-2 rounded-xl border border-gray-200 dark:border-gray-700 shadow-sm">
               <button onClick={lang === 'ar' ? handleNextWeek : handlePrevWeek} disabled={lang === 'ar' ? currentWeek === planData.totalWeeks : currentWeek === 1} className="p-1.5 rounded-lg hover:bg-gray-200 dark:hover:bg-gray-800 disabled:opacity-30 transition-colors">
                 {lang === 'ar' ? <ChevronRight className="w-5 h-5 text-gray-600 dark:text-gray-300" /> : <ChevronLeft className="w-5 h-5 text-gray-600 dark:text-gray-300" />}
               </button>
               <span className="font-bold text-blue-600 dark:text-blue-400 min-w-[120px] text-center text-lg">
                 {t.week} {currentWeek} <span className="text-sm opacity-60">/ {planData.totalWeeks}</span>
               </span>
               <button onClick={lang === 'ar' ? handlePrevWeek : handleNextWeek} disabled={lang === 'ar' ? currentWeek === 1 : currentWeek === planData.totalWeeks} className="p-1.5 rounded-lg hover:bg-gray-200 dark:hover:bg-gray-800 disabled:opacity-30 transition-colors">
                 {lang === 'ar' ? <ChevronLeft className="w-5 h-5 text-gray-600 dark:text-gray-300" /> : <ChevronRight className="w-5 h-5 text-gray-600 dark:text-gray-300" />}
               </button>
             </div>
             
             <div className="flex items-center gap-3">
                <button onClick={handlePrint} className="bg-blue-600 hover:bg-blue-700 text-white flex items-center gap-2 text-sm font-bold transition-colors px-4 py-2 rounded-lg shadow-sm">
                  <Printer className="w-4 h-4" /> {t.exportPdf}
                </button>
                <button onClick={() => setPlanData(null)} className="text-gray-500 hover:text-gray-900 dark:hover:text-white flex items-center gap-2 text-sm font-bold transition-colors bg-gray-100 dark:bg-gray-800 px-4 py-2 rounded-lg">
                  <RotateCcw className="w-4 h-4" /> {t.reset}
                </button>
             </div>
          </div>
          
          <div className="hidden print:block text-center mb-6">
             <h1 className="text-3xl font-bold text-black">{lang === 'ar' ? 'الجدول الأسبوعي للمذاكرة - Venoxa AI' : 'Weekly Study Plan - Venoxa AI'}</h1>
             <p className="text-gray-600 mt-2">{t.week} {currentWeek} | {t.currentLvl}: {userLevel} | {t.targetLvl}: {targetLevel}</p>
          </div>

          <div className="overflow-x-auto pb-6 custom-scrollbar print:overflow-visible">
            <div className="min-w-[900px] flex flex-col gap-3 print:min-w-full">
              <div className="grid grid-cols-9 gap-3">
                 <div className="col-span-1 bg-blue-600 text-white rounded-xl flex items-center justify-center font-bold py-4 shadow-md print:shadow-none print:border print:border-gray-300 print:text-black print:bg-gray-100">
                   {t.activity}
                 </div>
                 {[1,2,3,4,5,6,7].map(d => (
                   <div key={d} className="col-span-1 bg-blue-600 text-white rounded-xl flex flex-col items-center justify-center font-bold py-2 shadow-md print:shadow-none print:border print:border-gray-300 print:text-black print:bg-gray-100">
                     <span className="text-sm">{t.day} {d}</span>
                     <span className="text-xs bg-blue-700 print:bg-transparent print:border print:border-gray-400 px-2 py-0.5 rounded-full mt-1">{userLevel}</span>
                   </div>
                 ))}
                 <div className="col-span-1 bg-blue-600 text-white rounded-xl flex flex-col items-center justify-center font-bold py-2 shadow-md text-center print:shadow-none print:border print:border-gray-300 print:text-black print:bg-gray-100">
                   <span className="text-sm">{t.week}</span>
                   <span className="text-xs opacity-80 mt-1">{t.summary}</span>
                 </div>
              </div>

              {planData.categories.map((cat, idx) => (
                 <div key={idx} className="grid grid-cols-9 gap-3">
                    <div className={`col-span-1 rounded-xl p-3 flex flex-col items-start justify-center ${cat.headerBg} ${cat.textColor} font-bold shadow-sm print:shadow-none print:border print:border-gray-200 print:bg-white print:text-black`}>
                      <div className="flex items-center gap-2 mb-1"><cat.icon className="w-4 h-4"/> {cat.name}</div>
                      <div className="text-[10px] opacity-70 font-normal">Daily practice</div>
                    </div>
                    {[0,1,2,3,4,5,6].map(d => {
                       const isInteractive = cat.id === 'vocab' || cat.id === 'grammar';
                       return (
                         <button 
                           key={d} 
                           onClick={() => {
                              if(cat.id === 'vocab') openVocabModal(d);
                              if(cat.id === 'grammar') openGrammarModal(d);
                           }}
                           className={`col-span-1 rounded-xl flex flex-col items-center justify-center p-2 transition-all ${isInteractive ? `cursor-pointer hover:scale-105 hover:shadow-md border ${cat.id === 'vocab' ? 'border-fuchsia-300 dark:border-fuchsia-500/50' : 'border-orange-300 dark:border-orange-500/50'}` : 'cursor-default border border-transparent'} ${cat.cellBg} ${cat.textColor} font-semibold shadow-sm print:shadow-none print:border print:border-gray-200 print:bg-white print:text-black`}
                         >
                            {cat.id === 'vocab' ? (
                              <>
                                <span className="text-sm font-bold">{wordsPerDay} {t.words}</span>
                                <span className="text-xs opacity-70 mt-1">{cat.mins}m</span>
                              </>
                            ) : cat.id === 'grammar' ? (
                              <>
                                <span className="text-[10px] font-bold text-center px-1 leading-tight line-clamp-2 overflow-hidden text-ellipsis w-full" dir="ltr" title={relevantGrammarTopics[((currentWeek - 1) * 7 + d) % relevantGrammarTopics.length]}>
                                   {relevantGrammarTopics[((currentWeek - 1) * 7 + d) % relevantGrammarTopics.length]}
                                </span>
                                <span className="text-[10px] opacity-70 mt-0.5">{cat.mins}m</span>
                              </>
                            ) : (
                              <span className="text-sm">{cat.mins}m</span>
                            )}
                         </button>
                       );
                    })}
                    <div className={`col-span-1 rounded-xl p-2 flex flex-col items-center justify-center ${cat.headerBg} ${cat.textColor} font-bold shadow-sm opacity-90 print:shadow-none print:border print:border-gray-200 print:bg-gray-50 print:text-black`}>
                       {formatTime(cat.mins * 7)}
                    </div>
                 </div>
              ))}

              <div className="grid grid-cols-9 gap-3 mt-2">
                <div className="col-span-1 rounded-xl p-3 flex flex-col items-start justify-center bg-gray-200 dark:bg-gray-800 text-gray-700 dark:text-gray-300 font-bold shadow-sm print:shadow-none print:border print:border-gray-300 print:bg-gray-100 print:text-black">
                  <div className="flex items-center gap-2"><Clock className="w-4 h-4"/> {t.categories.total}</div>
                </div>
                {[0,1,2,3,4,5,6].map(d => (
                   <div key={d} className="col-span-1 rounded-xl flex items-center justify-center p-2 bg-gray-100 dark:bg-gray-800/50 text-gray-700 dark:text-gray-400 font-bold shadow-sm print:shadow-none print:border print:border-gray-200 print:bg-white print:text-black">
                      {formatTime(dailyTotalMins)}
                   </div>
                ))}
                <div className="col-span-1 rounded-xl p-2 flex items-center justify-center bg-gray-300 dark:bg-gray-700 text-gray-900 dark:text-white font-bold shadow-sm print:shadow-none print:border print:border-gray-300 print:bg-gray-100 print:text-black">
                   {formatTime(weeklyTotalMins)}
                </div>
              </div>

            </div>
          </div>

          {vocabModal && (
            <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm animate-fadeIn print:hidden" onClick={() => setVocabModal(null)}>
              <div className="bg-white dark:bg-[#1a1a1e] rounded-3xl shadow-2xl max-w-sm w-full border border-gray-200 dark:border-gray-700 overflow-hidden transform transition-all scale-100" onClick={e => e.stopPropagation()}>
                <div className="bg-gradient-to-r from-fuchsia-600 to-purple-600 p-5 text-white flex justify-between items-center shadow-md">
                  <div>
                     <h3 className="font-bold text-xl flex items-center gap-2"><BookOpen className="w-5 h-5"/> {t.vocabHeader} ({t.day} {vocabModal.day})</h3>
                     <p className="text-sm opacity-90 mt-1">{t.learnTarget}</p>
                  </div>
                  <button onClick={() => setVocabModal(null)} className="p-1.5 hover:bg-white/20 rounded-xl transition-colors"><X className="w-6 h-6"/></button>
                </div>
                <div className="p-4 max-h-[60vh] overflow-y-auto space-y-3 custom-scrollbar bg-gray-50 dark:bg-[#121214]">
                  {vocabModal.words.length === 0 ? (
                     <p className="text-center text-gray-500 py-10">لا توجد كلمات متاحة، يرجى رفع ملف الكلمات.</p>
                  ) : (
                     vocabModal.words.map((w, i) => (
                       <div key={i} className="flex flex-col p-4 bg-white dark:bg-[#1a1a1e] rounded-2xl border border-gray-100 dark:border-gray-800 shadow-sm hover:shadow-md transition-shadow">
                          <div className="flex items-center gap-3 mb-1">
                            <span className="w-7 h-7 rounded-full bg-fuchsia-100 dark:bg-fuchsia-900/30 text-fuchsia-600 dark:text-fuchsia-400 flex items-center justify-center text-sm font-bold shrink-0">{i+1}</span>
                            <span className="font-extrabold text-lg text-gray-900 dark:text-gray-100" dir="ltr">{safeRender(w?.word)}</span>
                          </div>
                          {w?.meaning && <span className="text-sm text-gray-500 dark:text-gray-400 mr-10">{safeRender(w.meaning)}</span>}
                       </div>
                     ))
                  )}
                </div>
              </div>
            </div>
          )}

          {grammarModal && (
            <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm animate-fadeIn print:hidden" onClick={() => setGrammarModal(null)}>
              <div className="bg-white dark:bg-[#1a1a1e] rounded-3xl shadow-2xl max-w-sm w-full border border-gray-200 dark:border-gray-700 overflow-hidden transform transition-all scale-100" onClick={e => e.stopPropagation()}>
                <div className="bg-gradient-to-r from-orange-500 to-red-500 p-5 text-white flex justify-between items-center shadow-md">
                  <div>
                     <h3 className="font-bold text-xl flex items-center gap-2"><FileText className="w-5 h-5"/> {t.grammarHeader} ({t.day} {grammarModal.day})</h3>
                     <p className="text-sm opacity-90 mt-1">{t.grammarTarget}</p>
                  </div>
                  <button onClick={() => setGrammarModal(null)} className="p-1.5 hover:bg-white/20 rounded-xl transition-colors"><X className="w-6 h-6"/></button>
                </div>
                <div className="p-8 text-center bg-gray-50 dark:bg-[#121214]">
                   <div className="inline-flex p-4 rounded-full bg-orange-100 dark:bg-orange-500/20 text-orange-600 dark:text-orange-400 mb-6 shadow-sm">
                       <Target className="w-10 h-10" />
                   </div>
                   <h4 className="text-2xl font-extrabold text-gray-900 dark:text-white mb-4" dir="ltr">{safeRender(grammarModal.topic)}</h4>
                   <p className="text-sm text-gray-500 dark:text-gray-400 leading-relaxed">
                     {lang === 'ar' ? 'يمكنك التوجه لقسم "القواعد الشاملة" لطلب شرح مفصل لهذه القاعدة مع الأمثلة من معلم الذكاء الاصطناعي.' : 'Head over to the "Comprehensive Grammar" section to get a detailed AI explanation of this rule.'}
                   </p>
                </div>
              </div>
            </div>
          )}

        </div>
      )}
    </div>
  );
}

function Vocabulary({ AI_KEY, userLevel, lang, globalVocab, setGlobalVocab }) {
  const [srsData, setSrsData] = useState(() => JSON.parse(localStorage.getItem('venoxa_srs_data')) || {});
  
  const [vocabMode, setVocabMode] = useState('browse'); 
  const [listTab, setListTab] = useState('hard');
  const [tick, setTick] = useState(0); 

  useEffect(() => {
    const int = setInterval(() => setTick(t => t + 1), 60000);
    return () => clearInterval(int);
  }, []);
  
  const [levelFilter, setLevelFilter] = useState('All');
  const [searchQuery, setSearchQuery] = useState('');
  const [activeWordIndex, setActiveWordIndex] = useState(0);
  
  const [showMeaning, setShowMeaning] = useState(false);
  const [aiStory, setAiStory] = useState('');
  const [loadingAi, setLoadingAi] = useState(false);

  const [loadingContextStory, setLoadingContextStory] = useState(false);
  const [contextStory, setContextStory] = useState('');
  
  const [cardImage, setCardImage] = useState(null);
  const [loadingImage, setLoadingImage] = useState(false);
  
  const [uploadMessage, setUploadMessage] = useState('');

  const t = {
    browse: lang === 'ar' ? 'تصفح الكلمات' : 'Browse Words',
    review: lang === 'ar' ? 'المراجعة المتباعدة' : 'Spaced Review',
    lists: lang === 'ar' ? 'قوائم الكلمات' : 'Word Lists',
    hard: lang === 'ar' ? 'صعبة' : 'Hard',
    learning: lang === 'ar' ? 'قيد التعلم' : 'Learning',
    mastered: lang === 'ar' ? 'متقنة' : 'Mastered',
    emptyLists: lang === 'ar' ? 'لا توجد كلمات في هذه القائمة حالياً.' : 'No words in this list currently.',
    rateWordsMsg: lang === 'ar' ? 'قيم الكلمات في قسم "تصفح الكلمات" لتظهر هنا.' : 'Rate words in Browse mode to see them here.',
    contextBtn: lang === 'ar' ? 'اربط كل الكلمات الصعبة في قصة واحدة' : 'Link Hard Words in a Story',
    smartCards: lang === 'ar' ? 'البطاقات الذكية المخصصة' : 'Custom Smart Cards',
    totalWords: lang === 'ar' ? 'إجمالي الكلمات المتاحة:' : 'Total Available Words:',
    uploadCSV: lang === 'ar' ? 'رفع ملف (CSV)' : 'Upload CSV',
    searchWord: lang === 'ar' ? 'بحث عن كلمة' : 'Search Word',
    filterLevel: lang === 'ar' ? 'تصفية بالمستوى' : 'Filter by Level',
    manualSelect: lang === 'ar' ? 'اختيار يدوي' : 'Manual Select',
    noWords: lang === 'ar' ? 'لا توجد كلمات مطابقة' : 'No matching words',
    reviewDone: lang === 'ar' ? 'أنت رائع! 🎉' : 'Awesome! 🎉',
    reviewDoneMsg: lang === 'ar' ? 'لقد أنهيت جميع المراجعات المستحقة في هذا الوقت.' : 'You have completed all due reviews.',
    comeBackLater: lang === 'ar' ? 'عُد لاحقاً أو تصفح لتعلم المزيد.' : 'Come back later or browse to learn more.',
    browseLearn: lang === 'ar' ? 'تصفح وتعلم كلمات جديدة' : 'Browse and Learn New Words',
    noMatch: lang === 'ar' ? 'لا توجد كلمات مطابقة لبحثك.' : 'No words match your search.',
    clearFilters: lang === 'ar' ? 'مسح الفلاتر' : 'Clear Filters',
    storyBtn: lang === 'ar' ? 'قصة للربط الذهني' : 'Story for Memory',
    imgBtn: lang === 'ar' ? 'صورة توضيحية' : 'Illustrative Image',
    synonyms: lang === 'ar' ? 'مرادفات:' : 'Synonyms:',
    aiGenerated: lang === 'ar' ? 'مولدة بالذكاء الاصطناعي' : 'AI Generated',
    storyTitle: lang === 'ar' ? 'قصة ذكية للحفظ:' : 'Smart Memory Story:',
    clickToShow: lang === 'ar' ? 'انقر على البطاقة لإظهار المعنى والتقييم' : 'Click the card to show meaning and rate',
    rateAgain: lang === 'ar' ? 'أعد' : 'Again',
    rateHard: lang === 'ar' ? 'صعب' : 'Hard',
    rateGood: lang === 'ar' ? 'جيد' : 'Good',
    rateEasy: lang === 'ar' ? 'سهل' : 'Easy',
    time1m: lang === 'ar' ? '1 دقيقة' : '1 min',
    time10m: lang === 'ar' ? '10 دقائق' : '10 mins',
    time1d: lang === 'ar' ? '1 يوم' : '1 day',
    time4d: lang === 'ar' ? '4 أيام' : '4 days'
  };

  const dueCards = useMemo(() => {
    const now = Date.now();
    return globalVocab.filter(card => {
       const data = srsData[card.word];
       if (!data) return false;
       return data.nextReview <= now;
    });
  }, [globalVocab, srsData, tick]);

  const hardWords = useMemo(() => globalVocab.filter(w => srsData[w.word]?.rating === 'again' || srsData[w.word]?.rating === 'hard'), [globalVocab, srsData]);
  const learningWords = useMemo(() => globalVocab.filter(w => srsData[w.word]?.rating === 'good'), [globalVocab, srsData]);
  const masteredWords = useMemo(() => globalVocab.filter(w => srsData[w.word]?.rating === 'easy'), [globalVocab, srsData]);

  const currentListView = listTab === 'hard' ? hardWords : listTab === 'learning' ? learningWords : masteredWords;

  const parseCSVLine = (line) => {
      const result = [];
      let current = '';
      let inQuotes = false;
      for (let i = 0; i < line.length; i++) {
          const char = line[i];
          if (char === '"' && line[i+1] === '"') {
              current += '"';
              i++;
          } else if (char === '"') {
              inQuotes = !inQuotes;
          } else if (char === ',' && !inQuotes) {
              result.push(current.trim());
              current = '';
          } else {
              current += char;
          }
      }
      result.push(current.trim());
      return result;
  };

  const handleFileUpload = (event) => {
    const file = event.target.files[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (e) => {
      const text = e.target.result;
      const lines = text.split('\n');
      const parsedData = [];
      
      for (let i = 1; i < lines.length; i++) {
         if(!lines[i].trim()) continue;
         const rowArr = parseCSVLine(lines[i]);
         
         if(rowArr.length >= 4 && rowArr[0]) {
            parsedData.push({
               word: rowArr[0].replace(/^"|"$/g, ''),
               type: rowArr[1]?.replace(/^"|"$/g, '') || '',
               level: rowArr[2]?.replace(/^"|"$/g, '') || '',
               meaning: rowArr[3]?.replace(/^"|"$/g, '') || '',
               example: rowArr[4]?.replace(/^"|"$/g, '') || '',
               synonyms: rowArr[5]?.replace(/^"|"$/g, '') || ''
            });
         }
      }

      if(parsedData.length > 0) {
          setGlobalVocab(parsedData);
          setLevelFilter('All');
          setSearchQuery('');
          setVocabMode('browse');
          setUploadMessage(`تم رفع ${parsedData.length} كلمة بنجاح! ✅`);
          setTimeout(() => setUploadMessage(''), 3000);
      } else {
          setUploadMessage('لم يتم العثور على كلمات، تأكد من صيغة الملف ❌');
      }
    };
    reader.readAsText(file);
  };

  const filteredVocabList = useMemo(() => {
    return globalVocab.filter(item => {
      const matchesLevel = levelFilter === 'All' || item.level?.toUpperCase().includes(levelFilter.toUpperCase());
      const searchLower = searchQuery.toLowerCase();
      const matchesSearch = item.word.toLowerCase().includes(searchLower) || 
                            (item.meaning && item.meaning.toLowerCase().includes(searchLower));
      return matchesLevel && matchesSearch;
    });
  }, [globalVocab, levelFilter, searchQuery]);

  const safeIndex = activeWordIndex < filteredVocabList.length ? activeWordIndex : 0;
  
  const card = vocabMode === 'review' ? dueCards[0] : filteredVocabList[safeIndex];

  useEffect(() => {
    if (vocabMode === 'browse' && filteredVocabList.length > 0 && activeWordIndex >= filteredVocabList.length) {
      setActiveWordIndex(0);
    }
  }, [filteredVocabList.length, activeWordIndex, vocabMode]);

  const handleRate = (rating) => {
    if (!card) return;
    
    const now = Date.now();
    let nextReview = now;
    
    if (rating === 'again') nextReview = now + 1 * 60 * 1000;
    else if (rating === 'hard') nextReview = now + 10 * 60 * 1000;
    else if (rating === 'good') nextReview = now + 24 * 60 * 60 * 1000;
    else if (rating === 'easy') nextReview = now + 4 * 24 * 60 * 60 * 1000;

    const newData = { ...srsData, [card.word]: { nextReview, rating, lastReviewed: now } };
    setSrsData(newData);
    localStorage.setItem('venoxa_srs_data', JSON.stringify(newData));

    setShowMeaning(false);
    setAiStory('');
    setCardImage(null);
    
    if (vocabMode === 'browse') {
        setActiveWordIndex((safeIndex + 1) % filteredVocabList.length);
    }
  };

  const resetWordProgress = (wordString) => {
    const newData = { ...srsData };
    delete newData[wordString];
    setSrsData(newData);
    localStorage.setItem('venoxa_srs_data', JSON.stringify(newData));
  };

  const playAudio = (accentCode, wordToPlay = card?.word) => {
    if (!wordToPlay || !('speechSynthesis' in window)) return;
    window.speechSynthesis.cancel();
    const utterance = new SpeechSynthesisUtterance(wordToPlay);
    utterance.lang = accentCode;
    utterance.rate = 0.9;
    window.speechSynthesis.speak(utterance);
  };

  const generateStory = async () => {
    if(loadingAi || aiStory || !card) return;
    setLoadingAi(true);
    try {
      const prompt = `أنت معلم لغة إنجليزية مرح. قم بتأليف قصة قصيرة جداً ومضحكة (سطرين أو ثلاثة فقط) باللغة الإنجليزية تتضمن الكلمة "${card.word}" (والتي تعني: ${card.meaning}) لمساعدة طالب مستواه ${userLevel} على تذكرها بسهولة.
      بعد القصة الإنجليزية، أضف سطرين باللغة ${lang === 'ar' ? 'العربية' : 'الإنجليزية'} تشرح فيهما كيف تم استخدام الكلمة في القصة.`;
      const responseText = await fetchFromGemini(prompt, AI_KEY);
      const errJson = parseGeminiJSON(responseText);
      if (errJson && errJson.error) {
        setAiStory(errJson.error);
      } else {
        setAiStory(responseText);
      }
    } catch (err) {
      setAiStory(t.storyError);
    } finally {
      setLoadingAi(false);
    }
  };

  const generateImage = async () => {
    if(loadingImage || cardImage || !card) return;
    setLoadingImage(true);
    try {
      const promptText = `A clear, high-quality, and educational illustration representing the English word '${card.word}'. Context: ${card.meaning}. Make it visual and easy to understand without text. For example, if the word is 'across', show a person walking across a street.`;
      const url = `https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${AI_KEY}`;
      const payload = { instances: { prompt: promptText }, parameters: { sampleCount: 1 } };
      
      const response = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      
      if (!response.ok) throw new Error('Failed to generate image');
      const data = await response.json();
      const base64 = data.predictions?.[0]?.bytesBase64Encoded;
      
      if (base64) {
        setCardImage(`data:image/png;base64,${base64}`);
      } else {
        throw new Error('No image data returned');
      }
    } catch (err) {
      console.error("Image generation error", err);
      setCardImage(`https://image.pollinations.ai/prompt/${encodeURIComponent('A clear educational illustration representing the word ' + card.word + ' meaning ' + card.meaning)}?width=600&height=400&nologo=true`);
    } finally {
      setLoadingImage(false);
    }
  };

  const generateContextStory = async () => {
    if (hardWords.length === 0 || loadingContextStory) return;
    setLoadingContextStory(true);
    try {
      const wordsStr = hardWords.map(w => w.word).join(', ');
      const prompt = `You are a creative writing teacher. Use ALL of these words: [${wordsStr}] in a single short, coherent, and interesting story in English. The story should be accessible for a ${userLevel} learner. After the story, explain the context of each word in ${lang === 'ar' ? 'Arabic' : 'English'}.`;
      const response = await fetchFromGemini(prompt, AI_KEY);
      const errJson = parseGeminiJSON(response);
      if (errJson && errJson.error) {
        setContextStory(errJson.error);
      } else {
        setContextStory(response);
      }
    } catch (e) {
      setContextStory(t.storyError);
    } finally {
      setLoadingContextStory(false);
    }
  };

  return (
    <div className="max-w-3xl mx-auto space-y-6 animate-fadeIn">
      
      <div className="flex flex-col sm:flex-row bg-white dark:bg-[#1a1a1e] p-1.5 rounded-xl w-full max-w-2xl mx-auto mb-8 border border-gray-200 dark:border-gray-800 shadow-lg transition-colors">
        <button onClick={() => {setVocabMode('browse'); setShowMeaning(false); setCardImage(null); setAiStory('');}} className={`flex-1 py-2.5 rounded-lg font-bold text-sm transition-all flex justify-center items-center gap-2 ${vocabMode === 'browse' ? 'bg-blue-600 text-white shadow-md' : 'text-gray-600 dark:text-gray-400 hover:text-gray-900 dark:hover:text-white hover:bg-gray-50 dark:hover:bg-gray-800/50'}`}><Search className="w-4 h-4" /> {t.browse}</button>
        <button onClick={() => {setVocabMode('review'); setShowMeaning(false); setCardImage(null); setAiStory('');}} className={`flex-1 py-2.5 rounded-lg font-bold text-sm transition-all flex justify-center items-center gap-2 relative ${vocabMode === 'review' ? 'bg-gradient-to-r from-purple-600 to-blue-600 text-white shadow-md' : 'text-gray-600 dark:text-gray-400 hover:text-gray-900 dark:hover:text-white hover:bg-gray-50 dark:hover:bg-gray-800/50'}`}>
          <Layers className="w-4 h-4" /> {t.review}
          {dueCards.length > 0 && <span className="absolute -top-2 -right-2 bg-red-500 text-white text-xs font-bold w-5 h-5 flex items-center justify-center rounded-full animate-pulse shadow-lg">{dueCards.length}</span>}
        </button>
        <button onClick={() => {setVocabMode('lists'); setShowMeaning(false); setCardImage(null); setAiStory('');}} className={`flex-1 py-2.5 rounded-lg font-bold text-sm transition-all flex justify-center items-center gap-2 ${vocabMode === 'lists' ? 'bg-orange-500 text-white shadow-md' : 'text-gray-600 dark:text-gray-400 hover:text-gray-900 dark:hover:text-white hover:bg-gray-50 dark:hover:bg-gray-800/50'}`}><Archive className="w-4 h-4" /> {t.lists}</button>
      </div>

      {vocabMode === 'lists' && (
        <div className="bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-800 rounded-3xl p-6 md:p-8 shadow-2xl animate-fadeIn min-h-[500px] transition-colors">
          <div className="flex gap-2 mb-6 border-b border-gray-200 dark:border-gray-800 pb-4 overflow-x-auto no-scrollbar">
            <button onClick={() => setListTab('hard')} className={`px-4 py-2 rounded-lg font-bold whitespace-nowrap transition-colors ${listTab === 'hard' ? 'bg-red-50 dark:bg-red-500/20 text-red-600 dark:text-red-400 border border-red-200 dark:border-red-500/30' : 'text-gray-600 dark:text-gray-500 hover:text-gray-900 dark:hover:text-gray-300'}`}>{t.hard} ({hardWords.length})</button>
            <button onClick={() => setListTab('learning')} className={`px-4 py-2 rounded-lg font-bold whitespace-nowrap transition-colors ${listTab === 'learning' ? 'bg-blue-50 dark:bg-blue-500/20 text-blue-600 dark:text-blue-400 border border-blue-200 dark:border-blue-500/30' : 'text-gray-600 dark:text-gray-500 hover:text-gray-900 dark:hover:text-gray-300'}`}>{t.learning} ({learningWords.length})</button>
            <button onClick={() => setListTab('mastered')} className={`px-4 py-2 rounded-lg font-bold whitespace-nowrap transition-colors ${listTab === 'mastered' ? 'bg-green-50 dark:bg-green-500/20 text-green-600 dark:text-green-400 border border-green-200 dark:border-green-500/30' : 'text-gray-600 dark:text-gray-500 hover:text-gray-900 dark:hover:text-gray-300'}`}>{t.mastered} ({masteredWords.length})</button>
          </div>

          {listTab === 'hard' && hardWords.length > 0 && (
            <div className="mb-6">
              <button onClick={generateContextStory} disabled={loadingContextStory} className="w-full py-4 bg-orange-50 dark:bg-orange-600/10 text-orange-600 dark:text-orange-400 border border-orange-200 dark:border-orange-500/30 rounded-2xl font-bold flex items-center justify-center gap-2 hover:bg-orange-100 dark:hover:bg-orange-500/20 transition-all shadow-sm">
                {loadingContextStory ? <Activity className="w-5 h-5 animate-spin" /> : <><Sparkles className="w-5 h-5" /> ✨ {t.contextBtn}</>}
              </button>
              {contextStory && <div className={`mt-4 p-5 bg-gray-50 dark:bg-[#1a1a1e] rounded-2xl border border-gray-200 dark:border-gray-700 ${lang === 'ar' ? 'text-right' : 'text-left'} prose prose-orange dark:prose-invert max-w-none`}><div style={{ whiteSpace: 'pre-wrap' }}>{safeRender(contextStory)}</div></div>}
            </div>
          )}
          
          <div className="space-y-3 max-h-[500px] overflow-y-auto pr-2 custom-scrollbar">
            {currentListView.length === 0 ? (
               <div className="text-center text-gray-500 py-16 flex flex-col items-center"><Archive className="w-12 h-12 mb-4 opacity-20" /><p>{t.emptyLists}</p><p className="text-sm mt-2">{t.rateWordsMsg}</p></div>

function Grammar({ AI_KEY, userLevel, lang }) {
  const [openLevel, setOpenLevel] = useState(0);
  const [activeTopic, setActiveTopic] = useState(null);
  const [explanation, setExplanation] = useState("");
  const [loading, setLoading] = useState(false);

  const t = {
    title: lang === 'ar' ? 'القواعد الشاملة' : 'Comprehensive Grammar',
    subtitle: lang === 'ar' ? '(مع معلم الذكاء الاصطناعي)' : '(AI Teacher)',
    desc: lang === 'ar' ? 'مرتبة منهجياً من مستوى A1 إلى C1. اضغط على أي قاعدة ليقوم الذكاء الاصطناعي بشرحها لك.' : 'Structured from A1 to C1. Click any rule for an AI-generated explanation.',
    explainBtn: lang === 'ar' ? 'اشرح لي بالذكاء الاصطناعي' : 'Explain with AI',
    loading: lang === 'ar' ? 'جاري توليد الشرح والأمثلة...' : 'Generating explanation and examples...',
    error: lang === 'ar' ? 'عذراً، حدث خطأ.' : 'Error generating response.'
  };

  const handleExplain = async (topic, levelStr) => {
    if (activeTopic === topic) return; 
    setActiveTopic(topic);
    setExplanation("");
    setLoading(true);

    try {
      const prompt = `You are an expert English grammar teacher. Explain the grammar rule: "${topic}" for a ${levelStr} level student.
      The explanation MUST be in ${lang === 'ar' ? 'Arabic' : 'English'}. Keep it clear, simple, and provide 3 distinct examples in English with their translation in ${lang === 'ar' ? 'Arabic' : 'English'}.`;
      
      const responseText = await fetchFromGemini(prompt, AI_KEY);
      const errJson = parseGeminiJSON(responseText);
      if (errJson && errJson.error) {
        setExplanation(errJson.error);
      } else {
        setExplanation(responseText);
      }
    } catch (err) {
      setExplanation(t.error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-4xl mx-auto animate-fadeIn">
      <div className="mb-10 text-center">
         <h2 className="text-3xl font-extrabold text-gray-900 dark:text-white mb-3">{t.title}</h2>
         <p className="text-gray-600 dark:text-gray-400">{t.desc}</p>
      </div>
      <div className="space-y-4">
        {GRAMMAR_LEVELS.map((level, idx) => (
          <div key={idx} className="bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-800 rounded-2xl overflow-hidden transition-all duration-300 shadow-sm">
            <button onClick={() => { setOpenLevel(openLevel === idx ? -1 : idx); setActiveTopic(null); }} className={`w-full px-8 py-6 flex justify-between items-center text-xl font-bold transition-colors ${openLevel === idx ? 'bg-gray-50 dark:bg-[#1a1a1e] text-blue-600' : 'text-gray-800 dark:text-gray-200 hover:bg-gray-50'}`} dir="ltr">
              <div className="flex items-center gap-4"><div className={`w-3 h-3 rounded-full ${openLevel === idx ? 'bg-blue-500' : 'bg-gray-400 dark:bg-gray-600'}`}></div>{level.level}</div>
              <ChevronRight className={`w-6 h-6 transition-transform ${openLevel === idx ? 'rotate-90 text-blue-500' : 'text-gray-400'}`} />
            </button>
            <div className={`transition-all duration-500 overflow-hidden ${openLevel === idx ? 'max-h-[5000px] opacity-100' : 'max-h-0 opacity-0'}`}>
              <div className="p-8 bg-white dark:bg-[#121214] border-t border-gray-200 dark:border-gray-800 space-y-4">
                {level.topics.map((topic, tidx) => (
                  <div key={tidx} className="bg-gray-50 dark:bg-[#1a1a1e] rounded-xl border border-gray-200 dark:border-gray-800/50 overflow-hidden">
                    <div className="flex flex-col md:flex-row md:items-center justify-between p-4 gap-4">
                      <div className="flex items-center gap-3 text-lg text-gray-800 dark:text-gray-200" dir="ltr"><span className="font-semibold">{topic}</span></div>
                      <button onClick={() => handleExplain(topic, level.level)} className="px-4 py-2 rounded-lg text-sm font-bold flex items-center justify-center gap-2 bg-blue-50 text-blue-600"><Sparkles className="w-4 h-4" /> {t.explainBtn}</button>
                    </div>
                    {activeTopic === topic && (
                      <div className="bg-white dark:bg-[#0e0e11] border-t border-blue-100 p-6">
                        {loading ? <div className="flex items-center gap-3 text-blue-600"><Activity className="w-5 h-5 animate-spin" /> {t.loading}</div> : <div style={{ whiteSpace: 'pre-wrap' }}>{safeRender(explanation)}</div>}
                      </div>
                    )}
                  </div>
                ))}
              </div>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}

          </ul>
        </div>

        <div className="md:col-span-2 bg-white dark:bg-[#121214] p-8 rounded-3xl border border-gray-200 dark:border-gray-800 shadow-xl transition-colors">
          <h3 className="text-2xl font-bold text-gray-900 dark:text-white mb-8 border-b border-gray-200 dark:border-gray-800 pb-4">{lang === 'ar' ? 'المصادر' : 'Sources'}</h3>
          <div className="space-y-4 max-h-[400px] overflow-y-auto pr-2 custom-scrollbar">
            {currentLinks.map((link, idx) => (
              <div key={idx} className="bg-gray-50 dark:bg-[#1a1a1e] p-5 rounded-2xl border border-gray-200 dark:border-gray-800 transition-colors shadow-sm">
                <div className="text-left mb-3" dir="ltr">
                  <p className="font-bold text-gray-900 dark:text-gray-100 text-lg">{link.name}</p>
                  <span className="inline-block mt-1 text-xs font-bold text-blue-600 dark:text-blue-400 bg-blue-100 dark:bg-blue-500/10 px-2 py-1 rounded">{link.level}</span>
                </div>
                <a href={link.url} target="_blank" rel="noopener noreferrer" className="flex items-center justify-center w-full py-2 bg-gray-200 dark:bg-gray-800 hover:bg-gray-300 dark:hover:bg-gray-700 text-gray-800 dark:text-white rounded-xl text-sm font-bold transition-colors">
                  {lang === 'ar' ? 'تصفح المصدر' : 'Visit Source'}
                </a>
              </div>
            ))}
          </div>
        </div>
      </div>

    </div>
  );
}

function Writing({ AI_KEY, lang }) {
  const [text, setText] = useState('');
  const [feedback, setFeedback] = useState(null);
  const [loading, setLoading] = useState(false);

  const t = {
    title: lang === 'ar' ? 'مصحح الكتابة المتقدم' : 'Advanced AI Corrector',
    placeholder: lang === 'ar' ? "اكتب مقالاً قصيراً هنا (مثلاً عن طموحاتك)..." : "Write a short essay here...",
    btn: lang === 'ar' ? "تحليل وتصحيح النص ✨" : "Analyze & Correct Text ✨",
    corrected: lang === 'ar' ? "النص المصحح" : "Corrected Text",
    explanation: lang === 'ar' ? "الشرح:" : "Explanation:"
  };

  const handleCorrect = async () => {
    if (!text.trim()) return;
    setLoading(true);
    try {
      const prompt = `Correct this English text and provide a review. Respond strictly in JSON format: { "corrected_text": "...", "explanation": "Explanation in ${lang === 'ar' ? 'Arabic' : 'English'}", "score": "X/10" }. Text: "${text}"`;
      const raw = await fetchFromGemini(prompt, AI_KEY);
      const json = parseGeminiJSON(raw);
      
      if (json && json.error) {
        setFeedback({ error: json.error });
      } else {
        const safeRaw = raw || "";
        setFeedback(json || { 
          corrected_text: cleanFallbackText(safeRaw), 
          explanation: lang === 'ar' ? "مشكلة في التنسيق." : "Formatting issue.", 
          score: "" 
        });
      }
    } catch (e) { 
      setFeedback({ error: lang === 'ar' ? "خطأ في الاتصال بالذكاء الاصطناعي." : "Error connecting to AI." }); 
    }
    setLoading(false);
  };

  return (
    <div className="max-w-4xl mx-auto animate-fadeIn">
      <h2 className="text-3xl font-extrabold text-gray-900 dark:text-white mb-10 text-center flex items-center justify-center gap-3"><PenTool className="text-purple-500 dark:text-purple-400"/> {t.title}</h2>
      <textarea value={text} onChange={(e) => setText(e.target.value)} placeholder={t.placeholder} className="w-full h-64 bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-800 rounded-3xl p-8 text-gray-900 dark:text-white text-lg focus:outline-none focus:border-purple-500 transition-all mb-6 shadow-xl" dir="ltr"></textarea>
      <button onClick={handleCorrect} disabled={loading || !text.trim()} className="w-full bg-gradient-to-r from-purple-600 to-blue-600 text-white font-extrabold py-5 rounded-2xl shadow-lg hover:shadow-xl transition-all disabled:opacity-50">
        {loading ? <Activity className="animate-spin w-6 h-6 mx-auto" /> : t.btn}
      </button>
      
      {feedback && !feedback.error && (
        <div className="mt-10 space-y-6 animate-fadeIn">
          <div className="bg-green-50 dark:bg-[#121214] p-8 rounded-3xl border border-green-200 dark:border-green-500/20 shadow-sm">
            <h3 className="text-green-700 dark:text-green-400 font-bold mb-4 flex justify-between items-center text-xl">{t.corrected} <span className="text-yellow-600 dark:text-yellow-400">{safeRender(feedback.score)}</span></h3>
            <p className="text-gray-900 dark:text-white text-xl leading-relaxed" dir="ltr">{safeRender(feedback.corrected_text)}</p>
          </div>
          {feedback.explanation && (
            <div className="bg-white dark:bg-[#121214] p-8 rounded-3xl border border-gray-200 dark:border-gray-800 shadow-sm">
              <h3 className="text-blue-600 dark:text-blue-400 font-bold mb-2">{t.explanation}</h3>
              <p className="text-gray-700 dark:text-gray-400 leading-relaxed" style={{ whiteSpace: 'pre-wrap' }}>{safeRender(feedback.explanation)}</p>
            </div>
          )}
        </div>
      )}
      {feedback && feedback.error && (
        <div className="mt-6 bg-red-50 dark:bg-red-500/10 p-4 rounded-xl border border-red-200 dark:border-red-500/30 text-red-600 dark:text-red-400 text-center font-bold">
          {safeRender(feedback.error)}
        </div>
      )}
    </div>
  );
}

function Speaking({ AI_KEY, userLevel, lang }) {
  const t = {
    welcome: lang === 'ar' ? `مرحباً بك! بناءً على تقييمك، مهاراتك في التحدث ممتازة (${userLevel}). لنتدرب على بعض المواضيع المتقدمة.` : `Hello! Your speaking skills are evaluated at ${userLevel}. Let's practice some advanced topics.`,
    title: lang === 'ar' ? 'ممارسة طلاقة التحدث' : 'Fluency Practice',
    placeholder: lang === 'ar' ? 'اكتب رسالتك أو سجل صوتاً...' : 'Type or record your message...',
    thinking: lang === 'ar' ? 'الذكاء الاصطناعي يفكر...' : 'AI is thinking...',
    newChat: lang === 'ar' ? 'محادثة جديدة' : 'New Chat',
    callMode: lang === 'ar' ? 'مكالمة' : 'Call',
    roleplay: lang === 'ar' ? 'محاكاة (Roleplay)' : 'Roleplay Scenario',
    corrected: lang === 'ar' ? 'تم التصحيح التلقائي' : 'Auto-Corrected',
    clarity: lang === 'ar' ? 'الوضوح' : 'Clarity',
    grammar: lang === 'ar' ? 'القواعد' : 'Grammar',
    fluency: lang === 'ar' ? 'السلاسة' : 'Fluency',
    native: lang === 'ar' ? 'كيف يقولها المتحدث الأصلي؟' : 'How would a native say it?',
    listen: lang === 'ar' ? 'استمع للنطق الصحيح' : 'Listen to correct pronunciation',
    micOn: lang === 'ar' ? 'تحدث الآن، الميكروفون مفتوح...' : 'Speak now, microphone is open...',
    tapMic: lang === 'ar' ? 'اضغط للتحدث' : 'Tap to speak',
    callActive: lang === 'ar' ? 'مكالمة قيد الإجراء...' : 'Call in progress...',
    stopCall: lang === 'ar' ? 'إنهاء المكالمة' : 'End Call'
  };

  const [messages, setMessages] = useState([{ id: 1, role: 'ai', text: t.welcome }]);
  const [input, setInput] = useState('');
  const [loading, setLoading] = useState(false);
  const [chatLevel, setChatLevel] = useState(userLevel);
  const [topic, setTopic] = useState(SPEAKING_TOPICS[0]);
  const [isCallMode, setIsCallMode] = useState(false);
  const [isRecording, setIsRecording] = useState(false);
  const [expandedReviewId, setExpandedReviewId] = useState(null);
  const [loadingScenario, setLoadingScenario] = useState(false);

  const chatEndRef = useRef(null);
  const recognitionRef = useRef(null);

  useEffect(() => {
    chatEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages, expandedReviewId]);

  const speakText = (text) => {
    if ('speechSynthesis' in window) {
      window.speechSynthesis.cancel();
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.lang = 'en-US';
      utterance.rate = 0.95;
      window.speechSynthesis.speak(utterance);
    }
  };

  const handleSend = async (forcedText = null) => {
    const userMsg = forcedText || input;
    if (!userMsg.trim()) return;

    if (!forcedText) setInput('');
    const newMsgId = Date.now();
    setMessages(prev => [...prev, { id: newMsgId, role: 'user', text: userMsg }]);
    setLoading(true);

    try {
      const history = messages.map(m => `${m.role === 'ai' ? 'AI' : 'User'}: ${m.text}`).join('\n');
      
      const prompt = `You are a native English conversation partner and expert language tutor.
      User Level: ${chatLevel}. Topic: ${topic}.
      
      The user just said: "${userMsg}"
      Conversation history: ${history}
      
      Analyze the user's message and provide a conversational reply to keep the chat going.
      You MUST respond in valid JSON format ONLY. Do not use markdown wrappers. Format exactly like this:
      {
        "reply": "Your natural conversational reply in English.",
        "review": {
          "clarity": "Feedback on clarity in ${lang === 'ar' ? 'Arabic' : 'English'}.",
          "grammar": "Feedback on grammar in ${lang === 'ar' ? 'Arabic' : 'English'}.",
          "fluency": "Feedback on natural flow and fluency in ${lang === 'ar' ? 'Arabic' : 'English'}.",
          "native_way": "How a native speaker would express the exact same idea naturally in English."
        }
      }`;

      const rawResponse = await fetchFromGemini(prompt, AI_KEY);
      const jsonResponse = parseGeminiJSON(rawResponse);

      if (jsonResponse && jsonResponse.error) {
        setMessages(prev => [...prev, { id: Date.now() + 1, role: 'system', text: jsonResponse.error }]);
        if (isCallMode) speakText("Sorry, I am having trouble connecting.");
      } else if (jsonResponse && jsonResponse.reply) {
        setMessages(prev => {
          const updated = prev.map(msg => msg.id === newMsgId ? { ...msg, review: jsonResponse.review } : msg);
          return [...updated, { id: Date.now() + 1, role: 'ai', text: jsonResponse.reply }];
        });
        if (isCallMode) speakText(jsonResponse.reply);
      } else {
        const fallbackText = cleanFallbackText(rawResponse);
        setMessages(prev => [...prev, { id: Date.now() + 1, role: 'ai', text: fallbackText }]);
        if (isCallMode) speakText(fallbackText);
      }
    } catch (err) {
      setMessages(prev => [...prev, { id: Date.now() + 1, role: 'system', text: lang === 'ar' ? "فشل الاتصال بالذكاء الاصطناعي." : "Error connecting to AI." }]);
    } finally {
      setLoading(false);
    }
  };

  const startNewConversation = async () => {
    setLoading(true);
    setMessages([]);
    try {
      const prompt = `You are a native English conversation partner. Start a friendly conversation with a student at ${chatLevel} level about the topic: "${topic}". Ask an engaging opening question to start the chat. Return ONLY the text of your greeting, no JSON, no markdown.`;
      const response = await fetchFromGemini(prompt, AI_KEY);
      const errJson = parseGeminiJSON(response);
      if (errJson && errJson.error) {
         setMessages([{ id: Date.now(), role: 'system', text: errJson.error }]);
      } else {
         setMessages([{ id: Date.now(), role: 'ai', text: cleanFallbackText(response) }]);
      }
    } catch (e) {
      setMessages([{ id: Date.now(), role: 'system', text: lang === 'ar' ? "فشل الاتصال." : "Connection failed." }]);
    } finally {
      setLoading(false);
    }
  };

  const handleVoiceRecord = () => {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRecognition) {
      alert(lang === 'ar' ? "متصفحك لا يدعم تسجيل الصوت. استخدم Chrome." : "Browser doesn't support voice. Use Chrome.");
      return;
    }

    if (isRecording) {
      recognitionRef.current?.stop();
      setIsRecording(false);
      return;
    }

    const recognition = new SpeechRecognition();
    recognitionRef.current = recognition;
    recognition.lang = 'en-US';
    recognition.interimResults = false;

    recognition.onstart = () => setIsRecording(true);
    
    recognition.onresult = (event) => {
      const transcript = event.results[0][0].transcript;
      if (isCallMode) {
        handleSend(transcript); 
      } else {
        setInput(prev => prev + ' ' + transcript);
      }
    };
    
    recognition.onerror = () => {
      setIsRecording(false);
      if(!isCallMode) alert(lang === 'ar' ? "لم يتم التعرف على الصوت." : "Voice not recognized.");
    };
    
    recognition.onend = () => setIsRecording(false);
    recognition.start();
  };

  const toggleCallMode = () => {
    setIsCallMode(!isCallMode);
    if (!isCallMode) {
      speakText("Call started. What would you like to talk about?");
    } else {
      window.speechSynthesis.cancel();
      if(recognitionRef.current) recognitionRef.current.stop();
      setIsRecording(false);
    }
  };

  const generateScenario = async () => {
    setLoadingScenario(true);
    try {
      const prompt = `Generate a realistic and challenging roleplay scenario in English for a student at level ${chatLevel}. 
      Example: "You are returning a broken laptop to a store and the clerk is being difficult." 
      Provide the scenario in English and a short instruction in ${lang === 'ar' ? 'Arabic' : 'English'}. Return text only.`;
      const response = await fetchFromGemini(prompt, AI_KEY);
      const errJson = parseGeminiJSON(response);
      if (errJson && errJson.error) {
        setMessages(prev => [...prev, { id: Date.now(), role: 'system', text: errJson.error }]);
      } else {
        setMessages(prev => [...prev, { id: Date.now(), role: 'ai', text: response }]);
      }
    } catch (e) {
      console.error(e);
    } finally {
      setLoadingScenario(false);
    }
  };

  if (isCallMode) {
    return (
      <div className="max-w-md mx-auto h-[80vh] flex flex-col items-center justify-between bg-gradient-to-b from-blue-50 dark:from-[#121214] to-white dark:to-[#09090b] border border-blue-100 dark:border-gray-800 rounded-[3rem] p-8 shadow-2xl relative overflow-hidden animate-fadeIn transition-colors">
        <div className="absolute top-0 left-0 w-full h-full bg-blue-500/5 pointer-events-none"></div>
        
        <div className="text-center mt-10 relative z-10">
          <h2 className="text-2xl font-bold text-gray-900 dark:text-white mb-2">Venoxa AI</h2>
          <p className="text-blue-600 dark:text-blue-400 flex items-center justify-center gap-2">
            <span className="w-2 h-2 rounded-full bg-blue-600 dark:bg-blue-500 animate-pulse"></span> {t.callActive}
          </p>
          <p className="text-sm text-gray-500 mt-2">Level: {chatLevel} | {topic.split('(')[0]}</p>
        </div>

        <div className="relative flex items-center justify-center w-full my-12 z-10">
          <div className={`absolute w-48 h-48 rounded-full ${loading ? 'bg-blue-500/20 animate-ping' : isRecording ? 'bg-red-500/20 animate-ping' : 'bg-gray-200 dark:bg-gray-800/50'}`}></div>
          <div className={`absolute w-36 h-36 rounded-full ${loading ? 'bg-blue-500/30 animate-pulse' : isRecording ? 'bg-red-500/30 animate-pulse' : 'bg-gray-300 dark:bg-gray-700/50'}`}></div>
          <div className="w-24 h-24 bg-white dark:bg-[#1a1a1e] rounded-full z-10 flex items-center justify-center shadow-xl border border-gray-100 dark:border-gray-700">
             {loading ? <Activity className="w-10 h-10 text-blue-500 dark:text-blue-400 animate-spin" /> : 
              isRecording ? <Mic className="w-10 h-10 text-red-500 animate-pulse" /> :
              <Volume2 className="w-10 h-10 text-gray-400" />}
          </div>
        </div>

        <div className="w-full relative z-10 flex flex-col gap-6 items-center mb-10">
          <p className="text-gray-500 dark:text-gray-400 text-center text-sm h-8 font-medium">
            {loading ? t.thinking : isRecording ? t.micOn : t.tapMic}
          </p>
          <div className="flex items-center gap-8">
            <button onClick={handleVoiceRecord} className={`w-16 h-16 rounded-full flex items-center justify-center transition-all shadow-lg ${isRecording ? 'bg-red-100 dark:bg-red-500/20 text-red-600 dark:text-red-500 border border-red-300 dark:border-red-500/50' : 'bg-gray-100 dark:bg-gray-800 text-gray-600 dark:text-gray-300 hover:bg-gray-200 dark:hover:bg-gray-700'}`}>
              <Mic className="w-6 h-6" />
            </button>
            <button onClick={toggleCallMode} title={t.stopCall} className="w-20 h-20 rounded-full flex items-center justify-center bg-red-500 hover:bg-red-600 text-white shadow-[0_0_20px_rgba(239,68,68,0.4)] transition-all">
              <PhoneOff className="w-8 h-8" />
            </button>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className="max-w-5xl mx-auto h-[85vh] flex flex-col animate-fadeIn border border-gray-200 dark:border-gray-800 rounded-3xl overflow-hidden shadow-2xl bg-white dark:bg-[#09090b] transition-colors">
      <div className="bg-gray-50 dark:bg-[#121214] p-5 border-b border-gray-200 dark:border-gray-800 flex flex-col lg:flex-row justify-between items-center gap-4 relative z-10 transition-colors">
        <div className="flex items-center gap-4 w-full lg:w-auto">
           <div className="w-12 h-12 bg-emerald-100 dark:bg-emerald-500/10 rounded-xl flex items-center justify-center border border-emerald-200 dark:border-emerald-500/20 shrink-0">
              <MessageSquare className="w-6 h-6 text-emerald-600 dark:text-emerald-400" />
           </div>
           <div>
             <h2 className="text-xl font-bold text-gray-900 dark:text-white flex items-center gap-2">{t.title}</h2>
           </div>
        </div>

        <div className="flex flex-wrap items-center gap-3 w-full lg:w-auto">
          <select value={chatLevel} onChange={(e) => setChatLevel(e.target.value)} className="bg-white dark:bg-[#1a1a1e] text-gray-900 dark:text-white border border-gray-200 dark:border-gray-700 rounded-lg px-3 py-2 text-sm font-bold focus:outline-none focus:border-emerald-500 appearance-none shadow-sm">
            {['A1', 'A2', 'B1', 'B2', 'C1', 'C2'].map(lvl => <option key={lvl} value={lvl}>Level {lvl}</option>)}
          </select>
          <select value={topic} onChange={(e) => setTopic(e.target.value)} className="bg-white dark:bg-[#1a1a1e] text-gray-900 dark:text-white border border-gray-200 dark:border-gray-700 rounded-lg px-3 py-2 text-sm font-bold focus:outline-none focus:border-emerald-500 appearance-none max-w-[150px] truncate shadow-sm">
            {SPEAKING_TOPICS.map(topicName => <option key={topicName} value={topicName}>{topicName}</option>)}
          </select>
          
          <button onClick={startNewConversation} disabled={loading} className="bg-blue-50 dark:bg-blue-500/10 text-blue-600 dark:text-blue-400 border border-blue-200 dark:border-blue-500/30 hover:bg-blue-600 hover:text-white px-3 py-2 rounded-lg text-sm font-bold transition-all flex items-center gap-2 shadow-sm">
            {loading ? <Activity className="w-4 h-4 animate-spin" /> : <><RotateCcw className="w-4 h-4" /> <span className="hidden sm:inline">{t.newChat}</span></>}
          </button>
          
          <button onClick={generateScenario} disabled={loadingScenario} className="bg-emerald-50 dark:bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border border-emerald-200 dark:border-emerald-500/30 hover:bg-emerald-600 hover:text-white px-3 py-2 rounded-lg text-sm font-bold transition-all flex items-center gap-2 shadow-sm">
            {loadingScenario ? <Activity className="w-4 h-4 animate-spin" /> : <><Sparkles className="w-4 h-4" /> <span className="hidden sm:inline">{t.roleplay}</span></>}
          </button>
          
          <button onClick={toggleCallMode} className="bg-emerald-50 dark:bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border border-emerald-200 dark:border-emerald-500/30 hover:bg-emerald-600 hover:text-white px-3 py-2 rounded-lg text-sm font-bold transition-all flex items-center gap-2 shadow-sm">
            <Phone className="w-4 h-4" /> <span className="hidden sm:inline">{t.callMode}</span>
          </button>
        </div>
      </div>

      <div className="flex-1 p-6 overflow-y-auto space-y-6 bg-[url('data:image/svg+xml;base64,PHN2ZyB3aWR0aD0iMjAiIGhlaWdodD0iMjAiIHhtbG5zPSJodHRwOi8vd3d3LnczLm9yZy8yMDAwL3N2ZyI+PGNpcmNsZSBjeD0iMSIgY3k9IjEiIHI9IjEiIGZpbGw9InJnYmEoMjU1LDI1NSwyNTUsMC4wMykiLz48L3N2Zz4=')] dark:bg-[url('data:image/svg+xml;base64,PHN2ZyB3aWR0aD0iMjAiIGhlaWdodD0iMjAiIHhtbG5zPSJodHRwOi8vd3d3LnczLm9yZy8yMDAwL3N2ZyI+PGNpcmNsZSBjeD0iMSIgY3k9IjEiIHI9IjEiIGZpbGw9InJnYmEoMjU1LDI1NSwyNTUsMC4wMykiLz48L3N2Zz4=')] scroll-smooth">
        {messages.map((msg, idx) => (
          <div key={msg.id} className={`flex flex-col ${msg.role === 'user' ? 'items-end' : 'items-start'}`}>
            
            <div className={`max-w-[85%] md:max-w-[70%] p-5 rounded-2xl shadow-md relative ${
              msg.role === 'user' ? 'bg-gradient-to-br from-emerald-600 to-emerald-500 text-white rounded-br-none' : 
              msg.role === 'system' ? 'bg-red-50 dark:bg-red-500/10 border border-red-200 dark:border-red-500/30 text-red-600 dark:text-red-400 w-full text-center' :
              'bg-gray-100 dark:bg-[#1a1a1e] border border-gray-200 dark:border-gray-800 text-gray-800 dark:text-gray-200 rounded-bl-none'
            }`} dir="ltr">
              <p className="text-lg leading-relaxed">{safeRender(msg.text)}</p>
              
              {msg.role === 'ai' && (
                 <button onClick={() => speakText(msg.text)} className="mt-3 bg-white/50 dark:bg-gray-800/50 hover:bg-emerald-100 dark:hover:bg-emerald-500/20 text-gray-500 dark:text-gray-400 hover:text-emerald-600 dark:hover:text-emerald-400 p-2 rounded-lg transition-colors border border-transparent hover:border-emerald-200 dark:hover:border-emerald-500/30">
                   <Volume2 className="w-5 h-5" />
                 </button>
              )}
            </div>

            {msg.role === 'user' && msg.review && (
              <div className="mt-2 max-w-[85%] md:max-w-[70%] bg-white dark:bg-[#121214] border border-gray-200 dark:border-gray-800 rounded-2xl overflow-hidden shadow-lg animate-fadeIn transition-colors">
                <button 
                  onClick={() => setExpandedReviewId(expandedReviewId === msg.id ? null : msg.id)}
                  className="w-full px-4 py-3 bg-gray-50 dark:bg-[#1a1a1e] hover:bg-gray-100 dark:hover:bg-gray-800 flex justify-between items-center text-sm font-bold text-gray-700 dark:text-gray-300 transition-colors"
                >
                  <span className="flex items-center gap-2 text-emerald-600 dark:text-emerald-400"><CheckCircle2 className="w-4 h-4"/> {t.corrected}</span>
                  <ChevronRight className={`w-4 h-4 transition-transform ${expandedReviewId === msg.id ? 'rotate-90' : ''}`} />
                </button>
                
                {expandedReviewId === msg.id && (
                  <div className="p-4 space-y-4 border-t border-gray-200 dark:border-gray-800 bg-white dark:bg-[#121214]">
                    <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
                      <div className="bg-gray-50 dark:bg-[#1a1a1e] p-3 rounded-xl border border-gray-200 dark:border-gray-800 shadow-sm">
                        <p className="text-xs text-gray-500 mb-1 font-bold">{t.clarity}</p>
                        <p className="text-sm text-gray-800 dark:text-gray-200">{safeRender(msg.review.clarity)}</p>
                      </div>
                      <div className="bg-gray-50 dark:bg-[#1a1a1e] p-3 rounded-xl border border-gray-200 dark:border-gray-800 shadow-sm">
                        <p className="text-xs text-gray-500 mb-1 font-bold">{t.grammar}</p>
                        <p className="text-sm text-gray-800 dark:text-gray-200">{safeRender(msg.review.grammar)}</p>
                      </div>
                      <div className="bg-gray-50 dark:bg-[#1a1a1e] p-3 rounded-xl border border-gray-200 dark:border-gray-800 shadow-sm">
                        <p className="text-xs text-gray-500 mb-1 font-bold">{t.fluency}</p>
                        <p className="text-sm text-gray-800 dark:text-gray-200">{safeRender(msg.review.fluency)}</p>
                      </div>
                    </div>
                    
                    <div className="bg-gradient-to-r from-emerald-50 dark:from-emerald-500/10 to-blue-50 dark:to-blue-500/10 p-4 rounded-xl border border-emerald-100 dark:border-emerald-500/20 relative shadow-sm">
                       <h4 className="text-sm font-bold text-emerald-700 dark:text-emerald-400 mb-2 flex items-center gap-2">
                         <Target className="w-4 h-4"/> {t.native}
                       </h4>
                       <p className="text-lg text-gray-900 dark:text-white font-medium" dir="ltr">"{safeRender(msg.review.native_way)}"</p>
                       
                       <div className="mt-4 flex gap-3">
                         <button onClick={() => speakText(safeRender(msg.review.native_way))} className="bg-emerald-600 hover:bg-emerald-700 text-white px-4 py-2 rounded-lg text-sm font-bold flex items-center gap-2 transition shadow-md">
                           <Play className="w-4 h-4"/> {t.listen}
                         </button>
                       </div>
                    </div>
                  </div>
                )}
              </div>
            )}
          </div>
        ))}
        {loading && (
          <div className="flex justify-start">
             <div className="bg-gray-100 dark:bg-[#1a1a1e] border border-gray-200 dark:border-gray-800 p-5 rounded-2xl rounded-bl-none flex items-center gap-3 shadow-sm">
                <div className="w-2 h-2 bg-emerald-500 rounded-full animate-bounce"></div>
                <div className="w-2 h-2 bg-emerald-500 rounded-full animate-bounce" style={{animationDelay: '0.2s'}}></div>
                <div className="w-2 h-2 bg-emerald-500 rounded-full animate-bounce" style={{animationDelay: '0.4s'}}></div>
             </div>
          </div>
        )}
        <div ref={chatEndRef} />
      </div>

      <div className="bg-gray-50 dark:bg-[#121214] p-5 border-t border-gray-200 dark:border-gray-800 relative z-10 transition-colors">
        <div className="flex items-center gap-3 bg-white dark:bg-[#1a1a1e] border border-gray-300 dark:border-gray-700 rounded-full p-2 focus-within:border-emerald-500/50 focus-within:shadow-[0_0_15px_rgba(16,185,129,0.1)] transition-all relative shadow-sm">
          
          <button onClick={handleVoiceRecord} className={`p-3.5 rounded-full transition-colors flex items-center justify-center ${isRecording ? 'bg-red-100 dark:bg-red-500/20 text-red-600 dark:text-red-500 animate-pulse' : 'bg-gray-100 dark:bg-gray-800 hover:bg-gray-200 dark:hover:bg-gray-700 text-emerald-600 dark:text-emerald-400'}`} title={isRecording ? "Stop Recording" : "Record Voice"}>
            {isRecording ? <MicOff className="w-5 h-5" /> : <Mic className="w-5 h-5" />}
          </button>
          
          {isRecording && <span className="absolute right-16 text-xs text-red-500 dark:text-red-400 font-bold animate-pulse">Listening...</span>}
          
          <input 
            type="text" 
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyDown={(e) => e.key === 'Enter' && handleSend()}
            placeholder={isRecording ? "" : t.placeholder}
            className="flex-1 bg-transparent border-none text-gray-900 dark:text-white text-lg px-2 focus:outline-none focus:ring-0 placeholder-gray-400 dark:placeholder-gray-500 disabled:opacity-50"
            dir="ltr"
            disabled={isRecording}
          />
          <button onClick={() => handleSend()} disabled={loading || !input.trim()} className="p-3.5 bg-emerald-500 hover:bg-emerald-600 rounded-full text-white transition-colors disabled:opacity-50 shadow-md">
            <Send className="w-5 h-5 ml-[-2px]" />
          </button>
        </div>
      </div>
    </div>
  );
}
